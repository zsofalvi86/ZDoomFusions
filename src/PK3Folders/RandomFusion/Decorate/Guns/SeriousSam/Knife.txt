Actor PPE_SSKnife : PPE_SeriousSamWeaponBase
{
	+WEAPON.NOALERT
	Tag "$TAG_SS_KN"
	Weapon.SlotNumber 1
	Weapon.Kickback 0
	DamageType "SeriousKnife"
	States
	{
	Ready.Ready:
    	SSKN A 1 A_WeaponReady
    	Loop
  	Deselect:
		SSKN A 1 A_Lower
		TNT1 A 0 A_Lower
    	Loop
  	Select:
		SSKN A 1 A_Raise
		TNT1 A 0 A_Raise
    	Loop
  	Fire:
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_PlayWeaponSound("weapons/ss/knife/swing")
    	SSKN BC 2 // 100
		TNT1 A 0 A_FireBullets(0,0,1,(100),"PPE_SSKnifePuff",FBF_NORANDOM,72)
		SSKN DE 2
		SSKN F 21
		SSKN E 2
		TNT1 A 0 A_ReFire
		Goto Ready.Ready
  	Flash:
    	TNT1 A 1 A_Light1
		TNT1 AAAA 1 //A_SetPitch(pitch-0.4)
		TNT1 A 0 A_Light2
		TNT1 AA 1 //A_SetPitch(pitch+0.8)
		Goto LightDone
  	Spawn:
    	SSKN W -1
    	Stop
	}
}

Actor PPE_SSKnifePuff : BulletPuff
{
	+NOEXTREMEDEATH
	SeeSound ""
	AttackSound ""
	States
	{
	Spawn:
	Crash:
	Melee:
		TNT1 A 0
		PUFF AB 2 Bright
		PUFF CD 2
		Stop
  }
}
