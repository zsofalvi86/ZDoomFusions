Actor PPE_DoomShotgun : PPE_DoomWeaponBase
{
	Tag "$TAG_DOOM_SG"
	Inventory.PickupSound "weapons/doom/shotgun/pickup"
	Weapon.UpSound "weapons/doom/shotgun/up"
	Weapon.SelectionOrder 600
	Weapon.SlotPriority 4
	Weapon.SlotNumber 3
	Weapon.AmmoUse1 8
	Weapon.AmmoUse2 8
	Weapon.AmmoType1 "PPE_Shells"
	Weapon.AmmoType2 "PPE_Shells"
	Weapon.AmmoGive 80
	Decal "BulletChip"
	PPE_WeaponBase.SlotAlt1 "PPE_DoomShotgunS2"
	PPE_WeaponBase.SpawnerId1 0,1
	PPE_WeaponBase.SpawnerId2 15,1
	States
	{
	Ready.Ready:
		DSGG A 1 A_WeaponReady
    	Loop
	Deselect:
    	DSGG A 1 A_Lower
		TNT1 A 0 A_Lower
    	Loop
  	Select:
    	DSGG A 1 A_Raise
		TNT1 A 0 A_Raise
    	Loop
  	Fire:
		DSGG A 3
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_PlayWeaponSound("weapons/doom/shotgun/fire")
		TNT1 A 0 A_FireBullets(5.4,0.35,7,(10),"PPE_DoomBulletPuff",FBF_USEAMMO|FBF_NORANDOM)
	Anim:
		DSGF ABC 2 bright
		DSGG A 1
		TNT1 A 0 A_PlaySound("weapons/doom/shotgun/cock",5)
    	DSGG DEF 2
		TNT1 A 0 A_FireCustomMissile("PPE_DoomSGShellCasingShooter",0,0,0,6)
		DSGG GHIJIHG 2
		DSGG FED 2
		DSGG A 1
    	DSGG A 7 A_ReFire
    	Goto Ready.Ready
  	Flash:
    	TNT1 A 2 A_Light1
		TNT1 AAAAA 1 A_SetPitch(pitch-0.24)
		TNT1 A 0 A_Light2
		TNT1 AAA 1 A_SetPitch(pitch+0.4)
		Goto LightDone
  	Spawn:
    	DSGW A -1
    	Stop
	}
}
