Actor PPE_DoomPistol : PPE_DoomWeaponBase
{
	Tag "$TAG_DOOM_PS"
	Inventory.PickupSound "weapons/doom/pistol/pickup"
	Weapon.UpSound "weapons/doom/pistol/up"
	Weapon.SelectionOrder 1300
	Weapon.SlotNumber 2
	Weapon.SlotPriority 4
	Weapon.AmmoUse1 14
	Weapon.AmmoUse2 14
	Weapon.AmmoType1 "PPE_Bullets"
	Weapon.AmmoType2 "PPE_Bullets"
	Weapon.AmmoGive 200
	Decal "BulletChip"
	PPE_WeaponBase.SpawnerId1 16,1
	PPE_WeaponBase.SpawnerId2 16,2
	PPE_WeaponBaseBase.Tags "pistol"
	States
	{
	Ready.Ready:
		DPNG A 1 A_WeaponReady2
    	Loop
  	Fire:
		DPNG A 4
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_PlayWeaponSound("weapons/doom/pistol/fire")
		TNT1 A 0 A_FireBullets(5.4,0.35,1,(14),"PPE_DoomPistolBulletPuff",FBF_USEAMMO|FBF_NORANDOM)
		DPNF ABC 2 bright
		TNT1 A 0 A_FireCustomMissile("PPE_DoomPSBulletCasingShooter",0,0,0,6)
    	DPNG CDED 1
		TNT1 A 0 A_ReFire
		DPNG DBBAA 1
    	Goto Ready.Ready
  	Flash:
    	TNT1 A 1 A_Light1
		TNT1 AAA 1 A_SetPitch(pitch-0.24)
		TNT1 A 0 A_Light2
		TNT1 AA 1 A_SetPitch(pitch+0.36)
		Goto LightDone
  	Spawn:
    	DPUN A -1
    	Stop
	}
}

Actor PPE_DoomPistolBulletPuff : PPE_DoomBulletPuff
{
	PPE_MissileBase.Tags "humiliation"
}

Actor PPE_DoomPSBulletCasingShooter : PPE_CasingShooterBase
{
	Speed 11
	States
	{
	Spawn:
		TNT1 A 1
	Death:
		TNT1 A 0 A_SpawnProjectile("PPE_BulletCasing",0,0,FRandom(65,80),CMF_AIMDIRECTION|CMF_TRACKOWNER,FRandom(-12,-19))
		Stop
	}
}
