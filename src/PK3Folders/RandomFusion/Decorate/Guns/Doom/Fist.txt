Actor PPE_DoomFist : PPE_DoomWeaponBase
{
	Tag "$TAG_DOOM_FST"
	Weapon.SlotNumber 1
	Weapon.SlotPriority 2
	Decal "BulletChip"
	States
	{
	Ready.Ready:
    	DFKG A 1 A_WeaponReady
    	Loop
  	Deselect:
		DFKG A 1 A_Lower(12)
    	Loop
  	Select:
		DFKG A 1 A_Raise(12)
    	Loop
  	Fire:
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_PlayWeaponSound("weapons/doom/fist/swing")
		TNT1 A 0 A_PlaySound("items/quad/use",2,CountInv("PowerQuadDamage"))
		DFKG B 1 
		DFKG B 1 Offset(0,32)
		DFKG C 1 
		DFKG C 1 Offset(0,32)
		TNT1 A 0 A_JumpIf(CountInv("PowerStrength"),"Berserk")
		TNT1 A 0 A_FireBullets(0,0,1,(16),"PPE_DoomFistPuff",FBF_NORANDOM,64)
		Goto Anim
	Berserk:
		TNT1 A 0 A_FireBullets(0,0,1,(160),"PPE_DoomFistPuffBerserk",FBF_NORANDOM,64)
		Goto Anim
	Anim:
		DFKG D 1 Offset(-1,31)
		DFKG D 1 Offset(0,32)
		DFKG E 1 Offset(2,34)
		DFKG E 1 Offset(0,32)
		DFKG E 1 Offset(2,34)
		DFKG D 1 Offset(0,32)
		DFKG D 1 Offset(-1,31)
		DFKG C 1 Offset(0,32)
		DFKG C 1 
		DFKG B 1 
		DFKG B 1 
		DFKG A 1 
		DFKG A 1 
		TNT1 A 0 A_ReFire
		DFKG A 5
		Goto Ready.Ready
  	Spawn:
    	DFKW A -1
    	Stop
	}
}
