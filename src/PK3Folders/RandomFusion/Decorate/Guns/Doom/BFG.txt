Actor PPE_DoomBfg : PPE_DoomWeaponBase
{
	Tag "$TAG_DOOM_BFG"
	Inventory.PickupSound "weapons/doom/bfg/pickup"
	Weapon.UpSound "weapons/doom/bfg/up"
	Weapon.SelectionOrder 2600
	Weapon.SlotPriority 8
	Weapon.SlotNumber 0
	Weapon.AmmoUse1 40
	Weapon.AmmoUse2 40
	Weapon.AmmoType1 "PPE_Cells"
	Weapon.AmmoType2 "PPE_Cells"
	Weapon.AmmoGive 40
	PPE_WeaponBase.SlotAlt1 "PPE_DoomBfgS7"
	PPE_WeaponBase.SpawnerId1 5,1
	States
	{
	Ready.Ready:
		DBFG ABBCCDDEEFFGGHHIIAAAAAAAAAAAAAAAAAAAAA 1 A_WeaponReady
    	Loop
	Deselect:
    	DBFG A 1 A_Lower
		TNT1 A 0 A_Lower
    	Loop
  	Select:
    	DBFG A 1 A_Raise
		TNT1 A 0 A_Raise
    	Loop
  	Fire:
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_PlayWeaponSound("weapons/doom/bfg/fire")
		DBFC ABCD 2 bright
		DBFC EFGHIJ 2 bright
		DBFC EFGHI 2 bright
		TNT1 A 0 A_Jump(256,"Launch")
	Launch:
		TNT1 A 0 A_FireCustomMissile("PPE_DoomBfgBall",0,1,0,2,1,0)
		Goto Winddown
	Winddown:
		DBFF ABC 3 bright
		DBFR AB 2 bright
		TNT1 A 0 A_ReFire
		DBFR A 1 bright
		DBFR C 3
		DBFG A 16
    	Goto Ready.Ready
  	Flash:
    	TNT1 A 20 A_Light1
		TNT1 A 10 A_Light2
		TNT1 A 2
		TNT1 AAAAAA 1 A_SetPitch(pitch-0.4)
		TNT1 A 0 A_Light1
		TNT1 AAA 1 A_SetPitch(pitch+0.8)
		Goto LightDone
  	Spawn:
    	DBFW AAAAAABBCCDDCCBB 1
    	Loop
	}
}
