Actor PPE_DoomPlasmaGun : PPE_DoomWeaponBase
{
	Tag "$TAG_DOOM_PG"
	Inventory.PickupSound "weapons/doom/plasma/pickup"
	Weapon.SelectionOrder 750
	Weapon.SlotNumber 6
	Weapon.SlotPriority 9
	Weapon.AmmoUse 1
	Weapon.AmmoType "PPE_Cells"
	Weapon.AmmoGive 40
	Decal "BulletChip"
	PPE_WeaponBase.SlotAlt1 "PPE_DoomPlasmaGunS8"
	PPE_WeaponBase.SpawnerId1 4,1
	PPE_WeaponBaseBase.Tags "plasma"
	States
	{
	Ready.Ready:
    	DPSG ABBCCDDEEFFAAAAAAAAAAAAAAAAA 1 A_WeaponReady2
    	Loop
  	Fire:
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_PlayWeaponSound("weapons/doom/plasma/fire")
		TNT1 A 0 A_FireCustomMissile("PPE_DoomPlasma",0,1,0,2,1,0)
		Goto Anim1
	Anim1:
    	DPPF ABC 1 bright
		TNT1 A 0 A_ReFire("FireB")
		Goto Winddown
	FireB:
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_PlayWeaponSound("weapons/doom/plasma/fire")
		TNT1 A 0 A_FireCustomMissile("PPE_DoomPlasma",0,1,0,2,1,0)
		Goto Anim2
	Anim2:
		DPPF DEF 1 bright
		TNT1 A 0 A_ReFire("Fire")
	Winddown:
		DPPF G 1 bright
		DPSG A 1
		DPSR ABBBBA 3
		Goto Ready.Ready
  	Flash:
		TNT1 A 0 A_Light1
		TNT1 AA 1 A_SetPitch(pitch-0.14)
		TNT1 A 0 A_Light2
		TNT1 A 1 A_SetPitch(pitch+0.28)
		Goto LightDone
	Spawn:
    	DPLW AAAAAABBCCDDCCBB 1
		Loop
	}
}
