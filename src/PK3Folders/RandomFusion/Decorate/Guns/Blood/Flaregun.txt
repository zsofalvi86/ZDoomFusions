Actor PPE_BloodFlaregun : PPE_BloodWeaponBase
{
	Weapon.SlotNumber 1
	Weapon.UpSound "weapons/blood/flaregun/up"
	Tag "$TAG_BL_FG"
	Weapon.SlotNumber 2
	Weapon.AmmoUse1 6
	Weapon.AmmoUse2 48
	Weapon.AmmoType1 "PPE_Shells"
	Weapon.AmmoType2 "PPE_Shells"
	Weapon.AmmoGive 80
	PPE_WeaponBase.SlotAlt1 "PPE_BloodFlaregunS3"
	PPE_WeaponBase.SpawnerId1 0,1
	PPE_WeaponBase.SpawnerId1 0,3
	PPE_WeaponBase.SpawnerId1 1,2
	States
	{
	Ready.Ready:
		BFLG A 1 A_WeaponReady
		Loop
	Deselect:
		TNT1 A 0 A_Lower
		BFLG A 1 A_Lower
		Loop
	Select:
		TNT1 A 0 A_Raise
		BFLG A 1 A_Raise
		Loop
	Fire:
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_PlaySound("weapons/blood/flaregun/fire",1,1)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissile",0,1,0,2,1,0)
		BFLF ABC 2 bright
		BFLG CBA 2
		BFLG A 7
		TNT1 A 0 A_ReFire
		Goto Ready.Ready
	AltFire:
		TNT1 A 0 A_GunFlash
		TNT1 A 0 A_AlertMonsters
		TNT1 A 0 A_PlaySound("weapons/blood/flaregun/fire",1,1)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",-2.7,1,0,2,FPF_NOAUTOAIM ,-2) // mid flares
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",2.7,0,0,2,FPF_NOAUTOAIM ,2)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",-2.7,0,0,2,FPF_NOAUTOAIM ,2)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",2.7,0,0,2,FPF_NOAUTOAIM ,-2)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",-8,0,0,2,FPF_NOAUTOAIM ,0) // outer flares
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",8,0,0,2,FPF_NOAUTOAIM ,0)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",0,0,0,2,FPF_NOAUTOAIM ,6)
		TNT1 A 0 A_FireCustomMissile("PPE_BloodFlareMissileAlt",0,0,0,2,FPF_NOAUTOAIM ,-6)
		BFLF ABC 2 bright
		BFLG CBA 2
		BFLG A 10
		TNT1 A 0 A_ReFire
		Goto Ready.Ready
	 Flash:
    	TNT1 A 1 A_Light1
		TNT1 AAAAAA 1 A_SetPitch(pitch-0.25)
		TNT1 A 0 A_Light2
		TNT1 AAA 1 A_SetPitch(pitch+0.5)
		Goto LightDone
	Spawn:
		BFLW A -1
		Stop
  }
}
