Actor PPE_BloodPitchfork : PPE_BloodWeaponBase
{
	Weapon.SlotNumber 1
	Weapon.UpSound "weapons/blood/pitchfork/up"
	Tag "$TAG_BL_PF"
	States
	{
	Ready.Ready:
		PITF A 1 A_WeaponReady
		Loop
	Deselect:
		TNT1 A 0 A_Lower
		PITF A 1 A_Lower
		Loop
	Select:
		TNT1 A 0 A_Raise
		PITF A 1 A_Raise
		Loop
	Fire:
		TNT1 A 0 A_PlaySound("weapons/blood/pitchfork/swing",6,1)
		PITF B 1 offset(0,32)
		PITF B 1 offset(0,30)
		PITF B 1 offset(0,28)
		PITF C 1 offset(0,32)
		PITF C 1 offset(0,30)
		PITF C 1 offset(0,28)
		TNT1 A 0 A_FireCustomMissile("PPE_PitchforkMissile", 0, 0, 8, 6)
		TNT1 A 0 A_FireCustomMissile("PPE_PitchforkMissile", 0, 0, 3, 6)
		TNT1 A 0 A_FireCustomMissile("PPE_PitchforkMissile", 0, 0, -3, 6)
		TNT1 A 0 A_FireCustomMissile("PPE_PitchforkMissile", 0, 0, -8, 6)
		PITF B 1 offset(0,28)
		PITF B 1 offset(0,30)
		PITF B 1 offset(0,32)
		PITF A 6
		TNT1 A 0 A_ReFire
		Goto Ready.Ready
	Spawn:
		PITW A -1
		Stop
  }
}

		Stop
	}
}
