Class PPE_QuoteBase : Inventory abstract
{
	default {inventory.maxamount 1;}
	
	bool startedPlaying;
	
	override void AttachToOwner(Actor other)
	{
		startedPlaying = false;
		if (!other.FindInventory("PPE_QuoteBase",true) || bAlwaysPickup) //hijacking this flag for the force play
		{
			other.A_StartSound(PickupSound, CHAN_VOICE, CHANF_MAYBE_LOCAL, 1);//PlayPickupSound(other);
			startedPlaying = true;
			//other.A_Print("start_n: "..GetClassName());
		}
		super.AttachToOwner(other);
	}
	
	override void DoEffect()
	{
		if (!owner.IsActorPlayingSound(CHAN_VOICE,PickupSound))
		{
			if (!startedPlaying && bAdditiveTime) //hijacking this flag for the queueing
			{
				owner.A_StartSound(PickupSound, CHAN_VOICE, CHANF_MAYBE_LOCAL, 1);//PlayPickupSound(owner);
				startedPlaying = true;
				//owner.A_Print("start_q: "..GetClassName());
			}
			else
			{
				//owner.A_Print("stop: "..GetClassName());
				owner.SetInventory(GetClass(),0);
			}
		}
		Super.DoEffect();
	}
}
