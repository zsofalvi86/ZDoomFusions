Class PPE_GuardianHeadFireCircle : Actor
{
	mixin PPE_TimedLife;
	
	override void Tick()
	{
		tic++;
		if (tic >= tics)
		{
			self.bShootable = true;
			A_Die(deathState);
		}
		Super.Tick();
	}
	
	class<PPE_DotProvider> dotType;
	Property DotType: dotType;
	class<Actor> visualType;
	Property VisualType: visualType;
	int dotDamage;
	Property DotDamage: dotDamage;
	int period;
	Property Period: period;
	int periods;
	Property Periods: periods;
	
	Override int DoSpecialDamage(Actor target, int damage, name damagetype)
	{
		target.GiveInventory(dotType.GetClassName(),10);
		let inv = PPE_DotProvider(target.FindInventory(dotType.GetClassName()));
		if (inv) { inv.target = self.target; inv.Init(visualType, dotDamage, period, periods); }
		
		self.bShootable = true;
		A_Die("Normal");
		
		return Super.DoSpecialDamage(target,damage,damagetype);
	}
	
	Action void A_TraceFireball()
	{
		FLineTraceData trc;
		if (!target || !target.LineTrace(target.AngleTo(self),target.Distance3D(self)+200,0,0,32,0,0,trc)) {return;}
		if (trc.HitActor && trc.HitType==TRACE_HitActor && trc.HitActor.bShootable && !trc.HitActor.bKilled) {DoMissileDamage(trc.HitActor);}
	}
	
	Default
	{
		+NOBLOCKMAP;
		+NOGRAVITY;
		+NOTELEPORT;
		+CANNOTPUSH;
		+FORCEXYBILLBOARD;
		Radius 1;
		Height 1;
	}
}

Class PPE_SwOnFire : PPE_DotProvider
{
	int add;
	override void Init(class<Actor> visType, int dd, int prd, int prds)
	{
		visualType = visType;
		dotDamage = dd;
		period = prd;
		periods = prds;
		if (add < 1)
		{
			timers.Push(prds);
			add++;
		}
		else
		{
			timers[timers.Size()-1] += prds;
			add = 0;
		}
	}
	
	override void TimerExpired(int index)
	{
		owner.TakeInventory(GetClassName(),1);
	}
	
	override int GetDamage()
	{
		return dotDamage * timers.Size();
	}
}

	}
}
