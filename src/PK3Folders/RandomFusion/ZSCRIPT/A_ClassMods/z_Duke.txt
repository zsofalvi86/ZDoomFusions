Class PPE_DukeClassMod : PPE_ClassBase
{
	Default 
	{
		Inventory.PickupSound "weapons/dn/pickup"; ActiveSound "weapons/dn/pickup";
		Inventory.PickupMessage "$GOTCMODDUKE";
		PPE_ClassBase.SoundClass "PPE_DukePlayer";
	}
	
	override int GetSwapSpeed() { return 12; }
	
	int dmgbuffer;
	int lastquotetime;
	int lastkilltime;
	
	override void WorldThingDied (WorldEvent e)
	{
		if (e.Thing && e.Thing.Target && e.Thing.bIsMonster)
		{
			lastkilltime = level.time;
			if (e.Thing.GetSpawnHealth() > 1000)
			{
				e.Thing.Target.GiveInventory("PPE_DnBossQuote",1);
				lastquotetime = level.time;
			}
			else if (level.time > lastquotetime + 230)
			{
				if (e.Thing.health < -1*min(e.Thing.GetSpawnHealth(),150))
				{
					e.Thing.Target.GiveInventory("PPE_DnGibQuote",1);
					lastquotetime = level.time;
					//console.printf("%s", "gib");
				}
				else if (dmgbuffer >= 2500)
				{
					dmgbuffer = 0;
					e.Thing.Target.GiveInventory("PPE_DnDeathQuote",1);
					lastquotetime = level.time;
				}
				else
				{
					dmgbuffer += e.Thing.GetSpawnHealth() + 500 - min(500,(level.time-lastkilltime)*10);
				}
			}
		}
	}
	
	States
	{
	Spawn:
		CMDN ABCDEFG 2 bright;
		Loop;
	}
}
