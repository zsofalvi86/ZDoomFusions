Class PPE_BloodFlareAltOnFire : PPE_DotProvider
{
	override int GetDamage()
	{
		//console.printf("%i%s%s%s%i", owner.CountInv(GetClassName())," - ",GetClassName()," - ",singleDamage);
		if (owner) { return dotDamage * owner.CountInv(GetClassName()); }
		return dotDamage;
	}
	
	Default
	{
		Inventory.MaxAmount 100;
		
		+BLOODLESSIMPACT;
		+PAINLESS;

	}
}
